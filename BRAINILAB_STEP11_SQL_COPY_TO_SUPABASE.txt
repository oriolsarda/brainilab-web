-- BrainiLab Backend — Step 11: Admin / Backoffice V1
-- Run after Steps 1–10.
--
-- SECURITY MODEL
-- - Browser continues to use only the normal Supabase publishable key.
-- - service_role is NEVER used in admin JavaScript.
-- - every admin RPC re-checks auth.uid() against public.admin_users.
-- - optional per-admin MFA enforcement uses the JWT AAL claim.
-- - direct browser table access to admin/private operational tables is revoked.
-- - mutations write an immutable operational audit trail.
--
-- FIRST OWNER
-- After this migration, bootstrap the first owner MANUALLY in SQL Editor.
-- See SUPABASE_ADMIN_SETUP.md. Do not expose a bootstrap RPC to browsers.

begin;

create extension if not exists pgcrypto;


-- ============================================================
-- ADMIN IDENTITY / ROLES
-- ============================================================

create table if not exists public.admin_users (
  user_id uuid primary key
    references auth.users(id) on delete cascade,

  role text not null default 'support',
  active boolean not null default true,

  -- Keep false until TOTP/MFA has been enrolled for this admin.
  -- When true, all admin RPCs reject JWTs below aal2.
  require_mfa boolean not null default false,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),

  constraint admin_users_role_check
    check (role in ('owner','editor','support'))
);

create table if not exists public.admin_audit_log (
  id bigint generated by default as identity primary key,

  admin_user_id uuid not null
    references auth.users(id) on delete restrict,

  action text not null,
  entity_type text null,
  entity_id text null,

  metadata jsonb not null default '{}'::jsonb,

  created_at timestamptz not null default now(),

  constraint admin_audit_action_length
    check (char_length(action) between 3 and 100),

  constraint admin_audit_metadata_size
    check (octet_length(metadata::text) <= 20000)
);

create index if not exists admin_audit_created_idx
  on public.admin_audit_log(created_at desc);

create index if not exists admin_audit_entity_idx
  on public.admin_audit_log(entity_type,entity_id,created_at desc);

create or replace function public.require_brainilab_admin(
  p_roles text[] default array['owner','editor','support']::text[]
)
returns uuid
language plpgsql
stable
security definer
set search_path=public
as $$
declare
  v_uid uuid:=auth.uid();
  v_role text;
  v_require_mfa boolean:=false;
  v_aal text:=coalesce(auth.jwt()->>'aal','aal1');
begin
  if v_uid is null then
    raise exception 'Authentication required';
  end if;

  select
    au.role,
    au.require_mfa
  into
    v_role,
    v_require_mfa
  from public.admin_users au
  where au.user_id=v_uid
    and au.active=true;

  if v_role is null then
    raise exception 'Admin access required';
  end if;

  if not (v_role=any(p_roles)) then
    raise exception 'Admin permission denied';
  end if;

  if v_require_mfa and v_aal<>'aal2' then
    raise exception 'MFA required for this admin account';
  end if;

  return v_uid;
end;
$$;

revoke execute on function public.require_brainilab_admin(text[])
  from public,anon,authenticated;

create or replace function public.get_brainilab_admin_session()
returns jsonb
language plpgsql
stable
security definer
set search_path=public
as $$
declare
  v_uid uuid:=auth.uid();
  v_row record;
  v_profile record;
  v_aal text:=coalesce(auth.jwt()->>'aal','aal1');
begin
  if v_uid is null then
    return jsonb_build_object(
      'authenticated',false,
      'admin',false
    );
  end if;

  select *
    into v_row
  from public.admin_users
  where user_id=v_uid;

  if v_row.user_id is null or not v_row.active then
    return jsonb_build_object(
      'authenticated',true,
      'admin',false,
      'user_id',v_uid
    );
  end if;

  select
    p.display_name,
    p.avatar_url
  into v_profile
  from public.profiles p
  where p.user_id=v_uid;

  return jsonb_build_object(
    'authenticated',true,
    'admin',true,
    'user_id',v_uid,
    'role',v_row.role,
    'require_mfa',v_row.require_mfa,
    'aal',v_aal,
    'mfa_satisfied',not v_row.require_mfa or v_aal='aal2',
    'display_name',coalesce(v_profile.display_name,'BrainiLab Admin'),
    'avatar_url',v_profile.avatar_url,
    'permissions',case v_row.role
      when 'owner' then jsonb_build_array(
        'dashboard','daily','questions','content','results',
        'users','rankings','groups','suggestions','system','audit',
        'moderate_rankings','feature_flags'
      )
      when 'editor' then jsonb_build_array(
        'dashboard','daily','questions','content','results',
        'rankings','system','audit'
      )
      else jsonb_build_array(
        'dashboard','results','users','groups','suggestions','audit'
      )
    end
  );
end;
$$;

revoke execute on function public.get_brainilab_admin_session()
  from public,anon;

grant execute on function public.get_brainilab_admin_session()
  to authenticated;

create or replace function public.submit_brainilab_suggestion(
  p_type text,
  p_message text,
  p_email text default null,
  p_client_id text default null
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
  v_uid uuid:=auth.uid();
  v_type text:=lower(btrim(coalesce(p_type,'')));
  v_message text:=btrim(coalesce(p_message,''));
  v_email text:=lower(nullif(btrim(coalesce(p_email,'')),''));
  v_client text:=nullif(left(btrim(coalesce(p_client_id,'')),120),'');
  v_recent integer:=0;
  v_id uuid;
begin
  if v_type not in (
    'quiz_idea','game_idea','improvement','bug','general'
  ) then
    raise exception 'Invalid suggestion type';
  end if;

  if char_length(v_message) not between 4 and 4000 then
    raise exception 'Suggestion must be 4–4000 characters';
  end if;

  if v_email is not null
     and (
       char_length(v_email)>254
       or v_email !~ '^[^[:space:]@]+@[^[:space:]@]+\.[^[:space:]@]+$'
     ) then
    raise exception 'Invalid email';
  end if;

  -- Lightweight abuse guard. This is not a substitute for edge/CDN rate limiting.
  if v_uid is not null then
    select count(*)::integer
      into v_recent
    from public.suggestions s
    where s.user_id=v_uid
      and s.created_at>now()-interval '1 hour';
  elsif v_client is not null then
    select count(*)::integer
      into v_recent
    from public.suggestions s
    where s.client_id=v_client
      and s.created_at>now()-interval '1 hour';
  end if;

  if v_recent>=5 then
    raise exception 'Too many suggestions. Try again later.';
  end if;

  insert into public.suggestions(
    user_id,
    client_id,
    type,
    message,
    reply_email
  )
  values(
    v_uid,
    v_client,
    v_type,
    v_message,
    v_email
  )
  returning id into v_id;

  return jsonb_build_object(
    'ok',true,
    'suggestion_id',v_id
  );
end;
$$;

revoke execute on function public.submit_brainilab_suggestion(
  text,text,text,text
) from public;

grant execute on function public.submit_brainilab_suggestion(
  text,text,text,text
) to anon,authenticated;

create or replace function public.admin_set_brainilab_runtime_flag(
  p_flag_key text,
  p_enabled boolean,
  p_message text default null
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
  v_uid uuid;
begin
  v_uid:=public.require_brainilab_admin(
    array['owner']::text[]
  );

  if p_flag_key not in (
    'brainmix_enabled','flagdash_enabled','maphunt_enabled',
    'brainiword_enabled','rankings_enabled','groups_enabled',
    'maintenance_enabled'
  ) then
    raise exception 'Invalid runtime flag';
  end if;

  update public.runtime_flags
  set
    enabled=p_enabled,
    message=nullif(left(btrim(coalesce(p_message,'')),500),''),
    updated_by=v_uid,
    updated_at=now()
  where flag_key=p_flag_key;

  perform public.log_brainilab_admin_action(
    'RUNTIME_FLAG_UPDATED',
    'runtime_flag',
    p_flag_key,
    jsonb_build_object(
      'enabled',p_enabled,
      'message',nullif(left(btrim(coalesce(p_message,'')),500),'')
    )
  );

  return (
    select jsonb_build_object(
      'flag_key',rf.flag_key,
      'enabled',rf.enabled,
      'message',rf.message,
      'updated_at',rf.updated_at
    )
    from public.runtime_flags rf
    where rf.flag_key=p_flag_key
  );
end;
$$;

revoke execute on function public.admin_set_brainilab_runtime_flag(
  text,boolean,text
) from public,anon;

grant execute on function public.admin_set_brainilab_runtime_flag(
  text,boolean,text
) to authenticated;


-- ============================================================
-- RANKING MODERATION
-- Suspension removes an entity from public ranking surfaces only.
-- It does NOT modify scores, XP, streaks or game history.
-- ============================================================

create table if not exists public.admin_ranking_suspensions (
  entity_type text not null,
  entity_id uuid not null,

  reason text not null,
  active boolean not null default true,
  expires_at timestamptz null,

  created_by uuid not null
    references auth.users(id) on delete restrict,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),

  primary key(entity_type,entity_id),

  constraint admin_ranking_suspensions_type
    check (entity_type in ('user','group')),

  constraint admin_ranking_suspensions_reason
    check (char_length(btrim(reason)) between 4 and 1000)
);

create index if not exists admin_ranking_suspensions_active_idx
  on public.admin_ranking_suspensions(entity_type,active,expires_at);


create or replace function public.admin_set_ranking_suspension(
  p_entity_type text,
  p_entity_id uuid,
  p_active boolean,
  p_reason text,
  p_expires_at timestamptz default null
)
returns void
language plpgsql
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_type text:=lower(btrim(coalesce(p_entity_type,'')));
  v_reason text:=btrim(coalesce(p_reason,''));
begin
  v_uid:=public.require_brainilab_admin(
    array['owner']::text[]
  );

  if v_type not in ('user','group') then
    raise exception 'Invalid ranking entity type';
  end if;

  if char_length(v_reason) not between 4 and 1000 then
    raise exception 'Moderation reason must be 4–1000 characters';
  end if;

  insert into public.admin_ranking_suspensions(
    entity_type,
    entity_id,
    reason,
    active,
    expires_at,
    created_by,
    updated_at
  )
  values(
    v_type,
    p_entity_id,
    v_reason,
    p_active,
    p_expires_at,
    v_uid,
    now()
  )
  on conflict(entity_type,entity_id)
  do update set
    reason=excluded.reason,
    active=excluded.active,
    expires_at=excluded.expires_at,
    created_by=excluded.created_by,
    updated_at=now();

  perform public.log_brainilab_admin_action(
    case when p_active
      then 'RANKING_ENTITY_SUSPENDED'
      else 'RANKING_ENTITY_RESTORED'
    end,
    v_type,
    p_entity_id::text,
    jsonb_build_object(
      'reason',v_reason,
      'expires_at',p_expires_at
    )
  );
end;
$$;

revoke execute on function public.admin_set_ranking_suspension(
  text,uuid,boolean,text,timestamptz
) from public,anon;

grant execute on function public.admin_set_ranking_suspension(
  text,uuid,boolean,text,timestamptz
) to authenticated;

create or replace function public.record_brainilab_verified_question_answers(
  p_client_result_id text,
  p_context_type text,
  p_context_id uuid,
  p_answers jsonb
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
  v_uid uuid:=auth.uid();
  v_context text:=lower(btrim(coalesce(p_context_type,'')));

  v_session_id uuid;
  v_result_id uuid;
  v_total integer;

  v_item jsonb;
  v_qv uuid;
  v_selected uuid;
  v_is_correct boolean;
  v_response integer;
  v_count integer:=0;
begin
  if v_uid is null then
    raise exception 'Authentication required';
  end if;

  if v_context not in ('quiz_pack','daily') then
    raise exception 'Invalid analytics context';
  end if;

  if jsonb_typeof(coalesce(p_answers,'[]'::jsonb))<>'array' then
    raise exception 'Answers must be an array';
  end if;

  select
    gs.id,
    gr.id,
    coalesce(gr.verified_total_questions,gr.total_questions)
  into
    v_session_id,
    v_result_id,
    v_total
  from public.game_sessions gs
  join public.game_results gr
    on gr.session_id=gs.id
  where gs.user_id=v_uid
    and gs.client_result_id=p_client_result_id
    and gr.answers_verified=true
  limit 1;

  if v_result_id is null then
    raise exception 'Verified result not found';
  end if;

  if jsonb_array_length(p_answers)<>coalesce(v_total,0) then
    raise exception 'Analytics answer count does not match verified result';
  end if;

  if v_context='daily' then
    if not exists(
      select 1
      from public.game_sessions gs
      where gs.id=v_session_id
        and gs.daily_challenge_id=p_context_id
        and gs.game_id='brainmix'
    ) then
      raise exception 'Daily Brain Mix context does not match verified session';
    end if;
  else
    if not exists(
      select 1
      from public.quiz_packs qp
      join public.game_sessions gs
        on gs.id=v_session_id
      where qp.id=p_context_id
        and qp.status='published'
        and gs.game_id not in (
          'brainmix','flagdash','maphunt','brainiword'
        )
        and gs.set_number=qp.set_number
        and gs.difficulty=qp.difficulty
    ) then
      raise exception 'Quiz pack context does not match verified session';
    end if;

    update public.game_sessions
    set quiz_pack_id=coalesce(quiz_pack_id,p_context_id)
    where id=v_session_id
      and quiz_pack_id is null;

    if exists(
      select 1
      from public.game_sessions gs
      where gs.id=v_session_id
        and gs.quiz_pack_id<>p_context_id
    ) then
      raise exception 'Quiz pack context mismatch';
    end if;
  end if;

  for v_item in
    select value
    from jsonb_array_elements(p_answers)
  loop
    begin
      v_qv:=(v_item->>'question_version_id')::uuid;
    exception when others then
      raise exception 'Invalid question version ID';
    end;

    if v_context='daily' then
      if not exists(
        select 1
        from public.daily_challenge_questions dcq
        where dcq.daily_challenge_id=p_context_id
          and dcq.question_version_id=v_qv
      ) then
        raise exception 'Question does not belong to Daily context';
      end if;
    else
      if not exists(
        select 1
        from public.quiz_pack_questions qpq
        where qpq.quiz_pack_id=p_context_id
          and qpq.question_version_id=v_qv
      ) then
        raise exception 'Question does not belong to quiz pack';
      end if;
    end if;

    if nullif(v_item->>'selected_option_id','') is null then
      v_selected:=null;
      v_is_correct:=false;
    else
      begin
        v_selected:=(v_item->>'selected_option_id')::uuid;
      exception when others then
        raise exception 'Invalid selected option ID';
      end;

      select qo.is_correct
        into v_is_correct
      from public.question_options qo
      where qo.id=v_selected
        and qo.question_version_id=v_qv;

      if v_is_correct is null then
        raise exception 'Option does not belong to question';
      end if;
    end if;

    begin
      v_response:=nullif(v_item->>'response_time_ms','')::integer;
    exception when others then
      v_response:=null;
    end;

    insert into public.verified_question_answers(
      result_id,
      session_id,
      user_id,
      question_version_id,
      selected_option_id,
      is_correct,
      response_time_ms,
      context_type,
      context_id
    )
    values(
      v_result_id,
      v_session_id,
      v_uid,
      v_qv,
      v_selected,
      coalesce(v_is_correct,false),
      case when v_response is null then null else greatest(v_response,0) end,
      v_context,
      p_context_id
    )
    on conflict(result_id,question_version_id)
    do nothing;

    v_count:=v_count+1;
  end loop;

  return jsonb_build_object(
    'recorded',v_count,
    'result_id',v_result_id
  );
end;
$$;

revoke execute on function public.record_brainilab_verified_question_answers(
  text,text,uuid,jsonb
) from public,anon;

-- ============================================================
-- DAILY OPERATIONS / HEALTH
-- ============================================================

create or replace function public.admin_get_daily_health(
  p_date date default current_date
)
returns jsonb
language plpgsql
stable
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_daily record;
  v_sessions integer:=0;
  v_brainmix integer:=0;
  v_flags integer:=0;
  v_map integer:=0;
  v_word integer:=0;
  v_easy integer:=0;
  v_medium integer:=0;
  v_hard integer:=0;
begin
  v_uid:=public.require_brainilab_admin(
    array['owner','editor']::text[]
  );

  select *
    into v_daily
  from public.daily_challenges
  where challenge_date=p_date;

  if v_daily.id is null then
    return jsonb_build_object(
      'date',p_date,
      'exists',false,
      'healthy',false,
      'issues',jsonb_build_array('Daily challenge does not exist')
    );
  end if;

  select count(*)::integer into v_sessions
  from public.game_sessions gs
  where gs.daily_challenge_id=v_daily.id
    and gs.status='completed';

  select count(*)::integer into v_brainmix
  from public.daily_challenge_questions
  where daily_challenge_id=v_daily.id;

  select count(*)::integer into v_flags
  from public.daily_flag_dash_questions
  where daily_challenge_id=v_daily.id;

  select count(*)::integer into v_map
  from public.daily_map_hunt_questions
  where daily_challenge_id=v_daily.id;

  select count(*)::integer into v_word
  from public.daily_brainiword
  where daily_challenge_id=v_daily.id;

  select
    count(*) filter(where qv.difficulty='easy')::integer,
    count(*) filter(where qv.difficulty='medium')::integer,
    count(*) filter(where qv.difficulty='hard')::integer
  into
    v_easy,
    v_medium,
    v_hard
  from public.daily_challenge_questions dcq
  join public.question_versions qv
    on qv.id=dcq.question_version_id
  where dcq.daily_challenge_id=v_daily.id;

  return jsonb_build_object(
    'exists',true,
    'healthy',
      v_brainmix=10
      and v_easy=4
      and v_medium=4
      and v_hard=2
      and v_flags=30
      and v_map=10
      and v_word=1,

    'id',v_daily.id,
    'date',v_daily.challenge_date,
    'daily_number',v_daily.daily_number,
    'status',v_daily.status,
    'generation_version',v_daily.generation_version,
    'generated_at',v_daily.generated_at,
    'published_at',v_daily.published_at,

    'completed_sessions',v_sessions,
    'content_locked',v_sessions>0,

    'brainmix',jsonb_build_object(
      'count',v_brainmix,
      'easy',v_easy,
      'medium',v_medium,
      'hard',v_hard,
      'questions',(
        select coalesce(
          jsonb_agg(
            jsonb_build_object(
              'position',dcq.position,
              'question_version_id',qv.id,
              'prompt',qv.prompt,
              'difficulty',qv.difficulty,
              'topic',t.slug
            )
            order by dcq.position
          ),
          '[]'::jsonb
        )
        from public.daily_challenge_questions dcq
        join public.question_versions qv
          on qv.id=dcq.question_version_id
        join public.topics t
          on t.id=qv.primary_topic_id
        where dcq.daily_challenge_id=v_daily.id
      )
    ),

    'flagdash',jsonb_build_object(
      'count',v_flags,
      'items',(
        select coalesce(
          jsonb_agg(
            jsonb_build_object(
              'position',fd.position,
              'country',c.country_name,
              'iso2',c.iso2
            )
            order by fd.position
          ),
          '[]'::jsonb
        )
        from public.daily_flag_dash_questions fd
        join public.daily_countries c
          on c.id=fd.country_id
        where fd.daily_challenge_id=v_daily.id
      )
    ),

    'maphunt',jsonb_build_object(
      'count',v_map,
      'items',(
        select coalesce(
          jsonb_agg(
            jsonb_build_object(
              'position',mhq.position,
              'clue',mh.clue,
              'country',c.country_name,
              'iso2',c.iso2
            )
            order by mhq.position
          ),
          '[]'::jsonb
        )
        from public.daily_map_hunt_questions mhq
        join public.map_hunt_clues mh
          on mh.id=mhq.clue_id
        join public.daily_countries c
          on c.id=mh.country_id
        where mhq.daily_challenge_id=v_daily.id
      )
    ),

    'brainiword',jsonb_build_object(
      'count',v_word,
      'word',(
        select bw.word
        from public.daily_brainiword dbw
        join public.brainiword_words bw
          on bw.id=dbw.word_id
        where dbw.daily_challenge_id=v_daily.id
        limit 1
      )
    )
  );
end;
$$;

revoke execute on function public.admin_get_daily_health(date)
  from public,anon;

grant execute on function public.admin_get_daily_health(date)
  to authenticated;

create or replace function public.admin_regenerate_brainilab_daily(
  p_date date
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_id uuid;
  v_sessions integer:=0;
  v_status text;
begin
  v_uid:=public.require_brainilab_admin(
    array['owner','editor']::text[]
  );

  if p_date<=current_date then
    raise exception
      'Admin V1 only regenerates future Daily content. Use a runtime kill switch for problems affecting today.';
  end if;

  select
    dc.id,
    dc.status
  into
    v_id,
    v_status
  from public.daily_challenges dc
  where dc.challenge_date=p_date
  for update;

  if v_id is not null then
    select count(*)::integer
      into v_sessions
    from public.game_sessions gs
    where gs.daily_challenge_id=v_id
      and gs.status='completed';

    if v_sessions>0 then
      raise exception
        'Daily content is locked because completed sessions already exist';
    end if;

    delete from public.daily_challenges
    where id=v_id;
  end if;

  v_id:=public.generate_brainilab_daily_challenge(p_date);

  -- Step 7 trigger should already generate the other Daily Games.
  perform public.ensure_brainilab_daily_games(v_id);

  if p_date<=current_date then
    update public.daily_challenges
    set
      status='published',
      published_at=coalesce(published_at,now())
    where id=v_id;
  end if;

  perform public.log_brainilab_admin_action(
    'DAILY_REGENERATED',
    'daily_challenge',
    v_id::text,
    jsonb_build_object(
      'date',p_date,
      'previous_status',v_status
    )
  );

  return public.admin_get_daily_health(p_date);
end;
$$;

revoke execute on function public.admin_regenerate_brainilab_daily(date)
  from public,anon;

grant execute on function public.admin_regenerate_brainilab_daily(date)
  to authenticated;

create or replace function public.admin_list_questions(
  p_search text default null,
  p_status text default null,
  p_difficulty text default null,
  p_topic_slug text default null,
  p_limit integer default 50,
  p_offset integer default 0
)
returns jsonb
language plpgsql
stable
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_search text:=lower(nullif(btrim(coalesce(p_search,'')),''));
  v_limit integer:=least(100,greatest(1,coalesce(p_limit,50)));
  v_offset integer:=greatest(0,coalesce(p_offset,0));
begin
  v_uid:=public.require_brainilab_admin(
    array['owner','editor']::text[]
  );

  return jsonb_build_object(
    'rows',(
      select coalesce(
        jsonb_agg(row_json order by created_at desc),
        '[]'::jsonb
      )
      from (
        select
          qv.created_at,
          jsonb_build_object(
            'question_id',q.id,
            'question_status',q.status,
            'external_key',q.external_key,

            'question_version_id',qv.id,
            'version',qv.version,
            'version_status',qv.status,

            'prompt',qv.prompt,
            'explanation',qv.explanation,
            'difficulty',qv.difficulty,

            'topic_slug',t.slug,
            'topic_name',t.name,

            'published_at',qv.published_at,

            'used_daily_count',(
              select count(*)
              from public.daily_challenge_questions dcq
              where dcq.question_version_id=qv.id
            ),

            'used_pack_count',(
              select count(*)
              from public.quiz_pack_questions qpq
              where qpq.question_version_id=qv.id
            ),

            'analytics_attempts',(
              select count(*)
              from public.verified_question_answers vqa
              where vqa.question_version_id=qv.id
            ),

            'analytics_accuracy',(
              select round(
                avg(case when vqa.is_correct then 100 else 0 end)::numeric,
                1
              )
              from public.verified_question_answers vqa
              where vqa.question_version_id=qv.id
            )
          ) as row_json
        from public.question_versions qv
        join public.questions q
          on q.id=qv.question_id
        join public.topics t
          on t.id=qv.primary_topic_id
        where
          (
            v_search is null
            or lower(qv.prompt) like '%'||v_search||'%'
            or lower(q.external_key) like '%'||v_search||'%'
          )
          and (
            p_status is null
            or p_status=''
            or qv.status=p_status
          )
          and (
            p_difficulty is null
            or p_difficulty=''
            or qv.difficulty=p_difficulty
          )
          and (
            p_topic_slug is null
            or p_topic_slug=''
            or t.slug=p_topic_slug
          )
        order by qv.created_at desc
        limit v_limit
        offset v_offset
      ) x
    ),

    'total',(
      select count(*)
      from public.question_versions qv
      join public.questions q
        on q.id=qv.question_id
      join public.topics t
        on t.id=qv.primary_topic_id
      where
        (
          v_search is null
          or lower(qv.prompt) like '%'||v_search||'%'
          or lower(q.external_key) like '%'||v_search||'%'
        )
        and (
          p_status is null
          or p_status=''
          or qv.status=p_status
        )
        and (
          p_difficulty is null
          or p_difficulty=''
          or qv.difficulty=p_difficulty
        )
        and (
          p_topic_slug is null
          or p_topic_slug=''
          or t.slug=p_topic_slug
        )
    )
  );
end;
$$;

create or replace function public.admin_save_question(
  p_question_version_id uuid default null,
  p_external_key text default null,
  p_prompt text default null,
  p_explanation text default '',
  p_difficulty text default null,
  p_topic_slug text default null,
  p_options jsonb default '[]'::jsonb,
  p_tags jsonb default '[]'::jsonb,
  p_source_url text default null,
  p_status text default 'draft'
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_question_id uuid;
  v_qv_id uuid:=p_question_version_id;
  v_external text:=nullif(btrim(coalesce(p_external_key,'')),'');
  v_prompt text:=btrim(coalesce(p_prompt,''));
  v_explanation text:=btrim(coalesce(p_explanation,''));
  v_difficulty text:=lower(btrim(coalesce(p_difficulty,'')));
  v_topic text:=lower(btrim(coalesce(p_topic_slug,'')));
  v_status text:=lower(btrim(coalesce(p_status,'draft')));
  v_topic_id uuid;
  v_existing_status text;
  v_item jsonb;
  v_position integer:=0;
  v_correct integer:=0;
  v_tag text;
  v_tag_id uuid;
begin
  v_uid:=public.require_brainilab_admin(
    array['owner','editor']::text[]
  );

  if char_length(v_prompt) not between 4 and 1000 then
    raise exception 'Question prompt must be 4–1000 characters';
  end if;

  if char_length(v_explanation)>3000 then
    raise exception 'Explanation is too long';
  end if;

  if v_difficulty not in ('easy','medium','hard') then
    raise exception 'Invalid difficulty';
  end if;

  if v_status not in ('draft','review','published') then
    raise exception 'Invalid question status';
  end if;

  select t.id into v_topic_id
  from public.topics t
  where t.slug=v_topic
    and t.is_active=true;

  if v_topic_id is null then
    raise exception 'Topic not found';
  end if;

  if jsonb_typeof(coalesce(p_options,'[]'::jsonb))<>'array'
     or jsonb_array_length(p_options)<>4 then
    raise exception 'Exactly 4 options are required';
  end if;

  if exists(
    select 1
    from jsonb_array_elements(p_options) x
    where char_length(btrim(coalesce(x.value->>'text','')))<1
  ) then
    raise exception 'All 4 options need text';
  end if;

  if (
    select count(distinct lower(btrim(x.value->>'text')))
    from jsonb_array_elements(p_options) x
  )<>4 then
    raise exception 'Answer options must be unique';
  end if;

  if v_status='published' and char_length(v_explanation)<4 then
    raise exception 'Published questions need an explanation';
  end if;

  select count(*)::integer into v_correct
  from jsonb_array_elements(p_options) x
  where coalesce((x.value->>'is_correct')::boolean,false)=true;

  if v_correct<>1 then
    raise exception 'Exactly one option must be correct';
  end if;

  if p_question_version_id is null then
    if v_external is null then
      v_external:='admin-'||
        substr(replace(gen_random_uuid()::text,'-',''),1,16);
    end if;

    if char_length(v_external) not between 4 and 160 then
      raise exception 'External key must be 4–160 characters';
    end if;

    if exists(
      select 1 from public.questions q
      where q.external_key=v_external
    ) then
      raise exception 'External key already exists';
    end if;

    if exists(
      select 1
      from public.question_versions qv
      where lower(regexp_replace(btrim(qv.prompt),'\s+',' ','g'))
        =lower(regexp_replace(v_prompt,'\s+',' ','g'))
    ) then
      raise exception 'A very similar question already exists';
    end if;

    insert into public.questions(
      external_key,
      status
    )
    values(
      v_external,
      'active'
    )
    returning id into v_question_id;

    insert into public.question_versions(
      question_id,
      external_key,
      version,
      language,
      format,
      prompt,
      explanation,
      difficulty,
      primary_topic_id,
      source_url,
      fact_checked_at,
      status,
      published_at
    )
    values(
      v_question_id,
      v_external||':v1',
      1,
      'en',
      'multiple_choice',
      v_prompt,
      v_explanation,
      v_difficulty,
      v_topic_id,
      nullif(btrim(coalesce(p_source_url,'')),''),
      null,
      v_status,
      case when v_status='published' then now() else null end
    )
    returning id into v_qv_id;

  else
    select
      qv.question_id,
      qv.status
    into
      v_question_id,
      v_existing_status
    from public.question_versions qv
    where qv.id=p_question_version_id
    for update;

    if v_question_id is null then
      raise exception 'Question version not found';
    end if;

    if v_existing_status not in ('draft','review') then
      raise exception
        'Published question versions are immutable in Admin V1';
    end if;

    update public.question_versions
    set
      prompt=v_prompt,
      explanation=v_explanation,
      difficulty=v_difficulty,
      primary_topic_id=v_topic_id,
      source_url=nullif(btrim(coalesce(p_source_url,'')),''),
      fact_checked_at=fact_checked_at,
      status=v_status,
      published_at=case
        when v_status='published' then coalesce(published_at,now())
        else published_at
      end
    where id=v_qv_id;

    delete from public.question_options
    where question_version_id=v_qv_id;

    delete from public.question_tags
    where question_version_id=v_qv_id;
  end if;

  v_position:=0;

  for v_item in
    select value
    from jsonb_array_elements(p_options)
  loop
    v_position:=v_position+1;

    insert into public.question_options(
      question_version_id,
      position,
      option_text,
      is_correct
    )
    values(
      v_qv_id,
      v_position,
      btrim(v_item->>'text'),
      coalesce((v_item->>'is_correct')::boolean,false)
    );
  end loop;

  if jsonb_typeof(coalesce(p_tags,'[]'::jsonb))='array' then
    for v_item in
      select value
      from jsonb_array_elements(p_tags)
    loop
      v_tag:=lower(
        regexp_replace(
          btrim(v_item#>>'{}'),
          '[^a-z0-9]+',
          '-',
          'g'
        )
      );

      v_tag:=trim(both '-' from v_tag);

      if char_length(v_tag)>=2 then
        insert into public.tags(slug,name)
        values(
          v_tag,
          initcap(replace(v_tag,'-',' '))
        )
        on conflict(slug) do nothing;

        select id into v_tag_id
        from public.tags
        where slug=v_tag;

        insert into public.question_tags(
          question_version_id,
          tag_id
        )
        values(
          v_qv_id,
          v_tag_id
        )
        on conflict do nothing;
      end if;
    end loop;
  end if;

  perform public.log_brainilab_admin_action(
    case
      when p_question_version_id is null then 'QUESTION_CREATED'
      when v_status='published' then 'QUESTION_PUBLISHED'
      else 'QUESTION_DRAFT_UPDATED'
    end,
    'question_version',
    v_qv_id::text,
    jsonb_build_object(
      'status',v_status,
      'topic',v_topic,
      'difficulty',v_difficulty
    )
  );

  return public.admin_get_question(v_qv_id);
end;
$$;

create or replace function public.admin_preview_question_import(
  p_rows jsonb
)
returns jsonb
language plpgsql
stable
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_row jsonb;
  v_output jsonb:='[]'::jsonb;
  v_prompt text;
  v_external text;
  v_topic text;
  v_difficulty text;
  v_valid boolean;
  v_issues jsonb;
  v_duplicate boolean;
begin
  v_uid:=public.require_brainilab_admin(
    array['owner','editor']::text[]
  );

  if jsonb_typeof(coalesce(p_rows,'[]'::jsonb))<>'array' then
    raise exception 'Import rows must be an array';
  end if;

  if jsonb_array_length(p_rows)>500 then
    raise exception 'Import preview limit is 500 rows';
  end if;

  for v_row in
    select value
    from jsonb_array_elements(p_rows)
  loop
    v_prompt:=btrim(coalesce(v_row->>'prompt',''));
    v_external:=nullif(btrim(coalesce(v_row->>'external_key','')),'');
    v_topic:=lower(btrim(coalesce(v_row->>'topic_slug','')));
    v_difficulty:=lower(btrim(coalesce(v_row->>'difficulty','')));

    v_issues:='[]'::jsonb;

    if char_length(v_prompt)<4 then
      v_issues:=v_issues||jsonb_build_array('Prompt too short');
    end if;

    if v_difficulty not in ('easy','medium','hard') then
      v_issues:=v_issues||jsonb_build_array('Invalid difficulty');
    end if;

    if not exists(
      select 1
      from public.topics t
      where t.slug=v_topic
        and t.is_active=true
    ) then
      v_issues:=v_issues||jsonb_build_array('Topic not found');
    end if;

    if jsonb_typeof(coalesce(v_row->'options','[]'::jsonb))<>'array' then
      v_issues:=v_issues||jsonb_build_array('Options must be an array');
    elsif jsonb_array_length(v_row->'options')<>4 then
      v_issues:=v_issues||jsonb_build_array('Exactly 4 options required');
    else
      if exists(
        select 1
        from jsonb_array_elements(v_row->'options') x
        where char_length(btrim(coalesce(x.value->>'text','')))<1
      ) then
        v_issues:=v_issues||jsonb_build_array('All 4 options need text');
      elsif (
        select count(distinct lower(btrim(x.value->>'text')))
        from jsonb_array_elements(v_row->'options') x
      )<>4 then
        v_issues:=v_issues||jsonb_build_array('Answer options must be unique');
      end if;

      if (
        select count(*)
        from jsonb_array_elements(v_row->'options') x
        where lower(coalesce(x.value->>'is_correct','false'))='true'
      )<>1 then
        v_issues:=v_issues||jsonb_build_array('Exactly 1 correct option required');
      end if;
    end if;

    v_duplicate:=exists(
      select 1
      from public.question_versions qv
      where lower(regexp_replace(btrim(qv.prompt),'\s+',' ','g'))
        =lower(regexp_replace(v_prompt,'\s+',' ','g'))
    ) or (
      v_external is not null
      and exists(
        select 1
        from public.questions q
        where q.external_key=v_external
      )
    );

    if v_duplicate then
      v_issues:=v_issues||jsonb_build_array('Possible duplicate');
    end if;

    v_valid:=jsonb_array_length(v_issues)=0;

    v_output:=v_output||jsonb_build_array(
      jsonb_build_object(
        'row',v_row,
        'valid',v_valid,
        'duplicate',v_duplicate,
        'issues',v_issues
      )
    );
  end loop;

  return jsonb_build_object(
    'rows',v_output,
    'total',jsonb_array_length(p_rows),
    'valid',(
      select count(*)
      from jsonb_array_elements(v_output) x
      where (x.value->>'valid')::boolean=true
    )
  );
end;
$$;

create or replace function public.admin_generate_quiz_pack(
  p_topic_slug text,
  p_difficulty text,
  p_set_number integer,
  p_title text default null
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_topic_slug text:=lower(btrim(coalesce(p_topic_slug,'')));
  v_difficulty text:=lower(btrim(coalesce(p_difficulty,'')));
  v_topic_id uuid;
  v_topic_name text;
  v_set integer:=coalesce(p_set_number,0);
  v_version integer;
  v_title text;
  v_external text;
  v_pack_id uuid;
  v_count integer;
begin
  v_uid:=public.require_brainilab_admin(
    array['owner','editor']::text[]
  );

  if v_difficulty not in ('easy','medium','hard') then
    raise exception 'Invalid difficulty';
  end if;

  if v_set<=0 then
    raise exception 'Set number must be positive';
  end if;

  select t.id,t.name
    into v_topic_id,v_topic_name
  from public.topics t
  where t.slug=v_topic_slug
    and t.is_active=true;

  if v_topic_id is null then
    raise exception 'Topic not found';
  end if;

  select count(*)::integer
    into v_count
  from public.question_versions qv
  join public.questions q
    on q.id=qv.question_id
  where q.status='active'
    and qv.status='published'
    and qv.primary_topic_id=v_topic_id
    and qv.difficulty=v_difficulty;

  if v_count<20 then
    raise exception
      'Need at least 20 published % / % questions. Available: %',
      v_topic_slug,
      v_difficulty,
      v_count;
  end if;

  if not exists(
    select 1
    from public.quiz_packs qp
    where qp.topic_id=v_topic_id
      and qp.difficulty=v_difficulty
      and qp.set_number=v_set
  ) then
    raise exception
      'Admin V1 versions existing quiz sets only. Add new set navigation before creating a brand-new set number.';
  end if;

  select coalesce(max(qp.version),0)+1
    into v_version
  from public.quiz_packs qp
  where qp.topic_id=v_topic_id
    and qp.difficulty=v_difficulty
    and qp.set_number=v_set;

  v_title:=coalesce(
    nullif(btrim(coalesce(p_title,'')),''),
    v_topic_name||' · '||initcap(v_difficulty)||' · Set '||v_set
  );

  v_external:=
    v_topic_slug||'.'||v_difficulty||'.set'||v_set||'.v'||v_version;

  insert into public.quiz_packs(
    external_key,
    topic_id,
    title,
    difficulty,
    set_number,
    version,
    status,
    total_questions
  )
  values(
    v_external,
    v_topic_id,
    left(v_title,120),
    v_difficulty,
    v_set,
    v_version,
    'draft',
    20
  )
  returning id into v_pack_id;

  insert into public.quiz_pack_questions(
    quiz_pack_id,
    question_version_id,
    position
  )
  select
    v_pack_id,
    candidate.id,
    row_number() over(
      order by
        candidate.pack_uses asc,
        candidate.daily_uses asc,
        candidate.created_at desc,
        candidate.id
    )::integer
  from (
    select
      qv.id,
      qv.created_at,
      (
        select count(*)
        from public.quiz_pack_questions old_qpq
        where old_qpq.question_version_id=qv.id
      ) as pack_uses,
      (
        select count(*)
        from public.daily_challenge_questions old_dcq
        where old_dcq.question_version_id=qv.id
      ) as daily_uses
    from public.question_versions qv
    join public.questions q
      on q.id=qv.question_id
    where q.status='active'
      and qv.status='published'
      and qv.primary_topic_id=v_topic_id
      and qv.difficulty=v_difficulty
    order by
      (
        select count(*)
        from public.quiz_pack_questions old_qpq
        where old_qpq.question_version_id=qv.id
      ) asc,
      (
        select count(*)
        from public.daily_challenge_questions old_dcq
        where old_dcq.question_version_id=qv.id
      ) asc,
      qv.created_at desc,
      qv.id
    limit 20
  ) candidate;

  perform public.log_brainilab_admin_action(
    'QUIZ_PACK_GENERATED',
    'quiz_pack',
    v_pack_id::text,
    jsonb_build_object(
      'topic',v_topic_slug,
      'difficulty',v_difficulty,
      'set_number',v_set,
      'version',v_version
    )
  );

  return public.admin_get_quiz_pack(v_pack_id);
end;
$$;

  text,text,integer,text
) from public,anon;

  text,text,integer,text

create or replace function public.admin_replace_quiz_pack_question(
  p_pack_id uuid,
  p_position integer,
  p_question_version_id uuid
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_pack record;
  v_question record;
begin
  v_uid:=public.require_brainilab_admin(
    array['owner','editor']::text[]
  );

  select
    qp.status,
    qp.topic_id,
    qp.difficulty
  into v_pack
  from public.quiz_packs qp
  where qp.id=p_pack_id
  for update;

  if v_pack.status is null then
    raise exception 'Quiz pack not found';
  end if;

  if v_pack.status not in ('draft','review') then
    raise exception 'Published quiz packs are immutable';
  end if;

  if p_position not between 1 and 20 then
    raise exception 'Pack position must be 1–20';
  end if;

  select
    qv.id,
    qv.primary_topic_id,
    qv.difficulty,
    qv.status,
    q.status as question_status
  into v_question
  from public.question_versions qv
  join public.questions q
    on q.id=qv.question_id
  where qv.id=p_question_version_id;

  if v_question.id is null
     or v_question.status<>'published'
     or v_question.question_status<>'active' then
    raise exception 'Replacement question must be active and published';
  end if;

  if v_question.primary_topic_id<>v_pack.topic_id
     or v_question.difficulty<>v_pack.difficulty then
    raise exception 'Replacement question must match pack topic and difficulty';
  end if;

  if exists(
    select 1
    from public.quiz_pack_questions qpq
    where qpq.quiz_pack_id=p_pack_id
      and qpq.question_version_id=p_question_version_id
      and qpq.position<>p_position
  ) then
    raise exception 'This question is already in the pack';
  end if;

  update public.quiz_pack_questions
  set question_version_id=p_question_version_id
  where quiz_pack_id=p_pack_id
    and position=p_position;

  if not found then
    raise exception 'Pack position not found';
  end if;

  perform public.log_brainilab_admin_action(
    'QUIZ_PACK_QUESTION_REPLACED',
    'quiz_pack',
    p_pack_id::text,
    jsonb_build_object(
      'position',p_position,
      'question_version_id',p_question_version_id
    )
  );

  return public.admin_get_quiz_pack(p_pack_id);
end;
$$;

revoke execute on function public.admin_replace_quiz_pack_question(
  uuid,integer,uuid
) from public,anon;

  uuid,integer,uuid
) to authenticated;


create or replace function public.admin_publish_quiz_pack(
  p_pack_id uuid
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$

  v_uid uuid;
  v_status text;
begin
  v_uid:=public.require_brainilab_admin(

  );

create or replace function public.admin_add_brainiword_word(

)

language plpgsql

set search_path=public
as $$

  v_uid uuid;

  );

  do update set is_active=true
  returning id into v_id;

  perform public.log_brainilab_admin_action(
    'BRAINIWORD_WORD_ADDED',
    'brainiword_word',
    v_id::text,
    jsonb_build_object('word',v_word)
  );

  return jsonb_build_object(
    'id',v_id,
    'word',v_word,
    'active',true
  );
end;
$$;

      into v_remaining

    raise exception 'BrainiWord word not found';
  end if;

  perform public.log_brainilab_admin_action(
    'BRAINIWORD_WORD_TOGGLED',
    'brainiword_word',
    p_word_id::text,
    jsonb_build_object('active',p_active)

create or replace function public.admin_list_results(
  p_search text default null,
  p_game_id text default null,
  p_verification text default null,
  p_date date default null,
  p_limit integer default 50,
  p_offset integer default 0
)
returns jsonb
language plpgsql
stable
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_search text:=lower(nullif(btrim(coalesce(p_search,'')),''));
  v_limit integer:=least(100,greatest(1,coalesce(p_limit,50)));
  v_offset integer:=greatest(0,coalesce(p_offset,0));
begin
  v_uid:=public.require_brainilab_admin();

  return jsonb_build_object(
    'rows',(
      select coalesce(
        jsonb_agg(row_json order by completed_at desc),
        '[]'::jsonb
      )
      from (
        select
          gs.completed_at,
          jsonb_build_object(
            'result_id',gr.id,
            'session_id',gs.id,
            'user_id',gs.user_id,

            'player',p.display_name,
            'friend_code',p.friend_code,

            'game_id',gs.game_id,
            'daily_number',gs.daily_number,
            'difficulty',gs.difficulty,
            'set_number',gs.set_number,

            'score',gr.score,
            'correct',coalesce(
              gr.verified_correct_answers,
              gr.correct_answers
            ),
            'total',coalesce(
              gr.verified_total_questions,
              gr.total_questions
            ),
            'accuracy',gr.accuracy,
            'duration_ms',gr.duration_ms,

            'server_verified',gr.server_verified,
            'answers_verified',gr.answers_verified,

            'completed_at',gs.completed_at
          ) as row_json
        from public.game_sessions gs
        join public.game_results gr
          on gr.session_id=gs.id
        join public.profiles p
          on p.user_id=gs.user_id
        where
          (
            v_search is null
            or lower(p.display_name) like '%'||v_search||'%'
            or lower(p.friend_code) like '%'||v_search||'%'
          )
          and (
            p_game_id is null
            or p_game_id=''
            or gs.game_id=p_game_id
          )
          and (
            p_date is null
            or (gs.completed_at at time zone 'UTC')::date=p_date
          )
          and (
            p_verification is null
            or p_verification=''
            or (
              p_verification='verified'
              and gr.answers_verified=true
            )
            or (
              p_verification='unverified'
              and gr.answers_verified=false
            )
          )
        order by gs.completed_at desc
        limit v_limit
        offset v_offset
      ) x
    )
  );
end;
$$;

            )

          ),

          jsonb_agg(
            jsonb_build_object(

              'prompt',qv.prompt,

        where vqa.result_id=gr.id

    )
    from public.game_results gr
    join public.game_sessions gs
      on gs.id=gr.session_id

            )
          ) as row_json
        from public.groups g
        join public.profiles p
          on p.user_id=g.owner_id
        where
          v_search is null

          or lower(p.display_name) like '%'||v_search||'%'
          or g.id::text=v_search
        order by g.created_at desc
        limit v_limit
        offset v_offset
      ) x
    )
  );
end;
$$;

-- ============================================================
-- SYSTEM HEALTH / AUDIT
-- ============================================================

create or replace function public.admin_get_system_health()
returns jsonb
language plpgsql
stable
security definer
set search_path=public
as $$
declare
  v_uid uuid;
  v_cron jsonb:=null;
  v_latest date;
  v_future integer:=0;
  v_flags jsonb;
begin
  v_uid:=public.require_brainilab_admin(
    array['owner','editor']::text[]
  );

  select max(dc.challenge_date)
    into v_latest
  from public.daily_challenges dc;

  select count(*)::integer
    into v_future
  from public.daily_challenges dc
  where dc.challenge_date>current_date
    and dc.status='ready';

  select public.get_brainilab_runtime_flags()
    into v_flags;

  if to_regclass('cron.job') is not null then
    begin
      execute $cron$
        select coalesce(
          jsonb_agg(
            jsonb_build_object(
              'jobid',j.jobid,
              'jobname',j.jobname,
              'schedule',j.schedule,
              'active',j.active,
              'command',j.command,
              'last_status',(
                select d.status
                from cron.job_run_details d
                where d.jobid=j.jobid
                order by d.start_time desc
                limit 1
              ),
              'last_start',(
                select d.start_time
                from cron.job_run_details d
                where d.jobid=j.jobid
                order by d.start_time desc
                limit 1
              ),
              'last_end',(
                select d.end_time
                from cron.job_run_details d
                where d.jobid=j.jobid
                order by d.start_time desc
                limit 1
              )
            )
            order by j.jobid
          ),
          '[]'::jsonb
        )
        from cron.job j
        where j.jobname='brainilab-daily-maintenance'
      $cron$
      into v_cron;
    exception when others then
      v_cron:=jsonb_build_array(
        jsonb_build_object(
          'error','Cron metadata could not be read'
        )
      );
    end;
  end if;

  return jsonb_build_object(
    'database',jsonb_build_object(
      'profiles',to_regclass('public.profiles') is not null,
      'game_results',to_regclass('public.game_results') is not null,
      'question_bank',to_regclass('public.question_versions') is not null,
      'daily',to_regclass('public.daily_challenges') is not null,
      'progression',to_regclass('public.player_progression') is not null,
      'friends',to_regclass('public.friendships') is not null,
      'groups',to_regclass('public.groups') is not null,
      'admin',to_regclass('public.admin_users') is not null
    ),

    'daily',jsonb_build_object(
      'latest_generated_date',v_latest,
      'future_ready_days',v_future,
      'lookahead_days',(
        select lookahead_days
        from public.daily_generation_settings
        where singleton=true
      ),
      'cooldown_days',(
        select cooldown_days
        from public.daily_generation_settings
        where singleton=true
      )
    ),

    'content_pools',jsonb_build_object(
      'active_brainiword_words',(
        select count(*) from public.brainiword_words
        where is_active=true
      ),
      'active_map_clues',(
        select count(*) from public.map_hunt_clues
        where is_active=true
      ),
      'active_countries',(
        select count(*) from public.daily_countries
        where is_active=true
      )
    ),

    'cron',coalesce(v_cron,'[]'::jsonb),
    'runtime_flags',v_flags,
    'generated_at',now()
  );
end;
$$;

-- ============================================================
-- RANKING RPC MODERATION FILTERS
-- Replace Step 10 public Individual ranking with suspension exclusion.
-- ============================================================

create or replace function public.get_brainilab_individual_rankings(
  p_region text default 'global',
  p_country_code text default null,
  p_period text default 'daily',
  p_game_id text default 'all',
  p_metric text default 'score',
  p_limit integer default 100
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
  v_me uuid:=auth.uid();
  v_region text:=lower(coalesce(p_region,'global'));
  v_country text:=upper(nullif(btrim(coalesce(p_country_code,'')),''));
  v_period text:=lower(coalesce(p_period,'daily'));
  v_game text:=lower(coalesce(p_game_id,'all'));
  v_metric text:=lower(coalesce(p_metric,'score'));
  v_limit integer:=least(100,greatest(10,coalesce(p_limit,100)));
  v_my_country text;
  v_my_enabled boolean:=false;
  v_my_public_name text;
  v_rows jsonb;
  v_user jsonb;
  v_total integer:=0;
  v_user_eligible boolean:=false;
begin
  if v_region not in ('global','country') then
    raise exception 'Invalid ranking region';
  end if;

  if v_period not in ('daily','weekly','monthly') then
    raise exception 'Invalid ranking period';
  end if;

  if v_metric not in ('score','streak') then
    raise exception 'Invalid ranking metric';
  end if;

  if v_me is not null then
    select
      p.country_code,
      p.leaderboard_enabled,
      p.leaderboard_display_name
    into
      v_my_country,
      v_my_enabled,
      v_my_public_name
    from public.profiles p
    where p.user_id=v_me;

    if v_country is null then
      v_country:=v_my_country;
    end if;
  end if;

  if v_region='country'
     and (v_country is null or v_country !~ '^[A-Z]{2}$') then
    return jsonb_build_object(
      'rows','[]'::jsonb,
      'user',null,
      'total_players',0,
      'metric_label',case
        when v_metric='streak' then 'Streak'
        when v_game='all' then 'Brain Score'
        when v_game in ('brainmix','flagdash','maphunt','brainiword')
          then 'Daily points'
        else 'Points'
      end,
      'leaderboard_enabled',v_my_enabled,
      'leaderboard_display_name',v_my_public_name,
      'user_eligible',false,
      'country_required',true,
      'my_country',v_my_country,
      'generated_at',now()
    );
  end if;

  with candidates as (
    select
      p.user_id,
      p.leaderboard_display_name as public_name,
      p.country_code,
      public.brainilab_player_rank_value(
        p.user_id,
        v_period,
        v_game,
        v_metric
      ) as rank_value
    from public.profiles p
    where p.leaderboard_enabled=true
      and p.leaderboard_display_name is not null
      and char_length(btrim(p.leaderboard_display_name))>=2
      and (
        v_region='global'
        or p.country_code=v_country
      )
      and not exists(
        select 1
        from public.admin_ranking_suspensions ars
        where ars.entity_type='user'
          and ars.entity_id=p.user_id
          and ars.active=true
          and (
            ars.expires_at is null
            or ars.expires_at>now()
          )
      )
  ),
  eligible as (
    select *
    from candidates
    where rank_value>0
  ),
  ranked as (
    select
      e.*,
      row_number() over(
        order by
          e.rank_value desc,
          lower(e.public_name),
          e.user_id
      ) as rank
    from eligible e
  )
  select
    coalesce(
      jsonb_agg(
        jsonb_build_object(
          'rank',r.rank,
          'name',r.public_name,
          'country',r.country_code,
          'avatar',upper(left(r.public_name,1)),
          'score',r.rank_value,
          'streak',case
            when v_metric='streak' then r.rank_value
            else (
              select coalesce(pp.current_streak,0)
              from public.player_progression pp
              where pp.user_id=r.user_id
            )
          end,
          'display_value',case
            when v_metric='streak'
              then r.rank_value::text||' days'
            else to_char(r.rank_value,'FM999G999G999G990')
          end,
          'is_me',r.user_id=v_me
        )
        order by r.rank
      ) filter(where r.rank<=v_limit),
      '[]'::jsonb
    ),
    count(*)::integer,
    (
      select jsonb_build_object(
        'rank',mine.rank,
        'name',mine.public_name,
        'country',mine.country_code,
        'avatar',upper(left(mine.public_name,1)),
        'score',mine.rank_value,
        'streak',case
          when v_metric='streak' then mine.rank_value
          else (
            select coalesce(pp.current_streak,0)
            from public.player_progression pp
            where pp.user_id=mine.user_id
          )
        end,
        'display_value',case
          when v_metric='streak'
            then mine.rank_value::text||' days'
          else to_char(mine.rank_value,'FM999G999G999G990')
        end,
        'is_me',true
      )
      from ranked mine
      where mine.user_id=v_me
      limit 1
    ),
    exists(
      select 1 from ranked mine
      where mine.user_id=v_me
    )
  into
    v_rows,
    v_total,
    v_user,
    v_user_eligible
  from ranked r;

  return jsonb_build_object(
    'rows',v_rows,
    'user',v_user,
    'total_players',v_total,
    'metric_label',case
      when v_metric='streak' then 'Streak'
      when v_game='all' then 'Brain Score'
      when v_game in ('brainmix','flagdash','maphunt','brainiword')
        then 'Daily points'
      else 'Points'
    end,
    'leaderboard_enabled',v_my_enabled,
    'leaderboard_display_name',v_my_public_name,
    'user_eligible',v_user_eligible,
    'region',v_region,
    'country',v_country,
    'my_country',v_my_country,
    'period',v_period,
    'game_id',v_game,
    'generated_at',now()
  );
end;
$$;

revoke execute on function public.get_brainilab_individual_rankings(
  text,text,text,text,text,integer
) from public;

-- Group ranking moderation: replace Step 10's RPC with the same
-- scoring semantics plus an active suspension filter.

create or replace function public.get_brainilab_group_rankings(
  p_region text default 'global',
  p_country_code text default null,
  p_period text default 'daily',
  p_game_id text default 'all',
  p_metric text default 'score',
  p_limit integer default 100
)
returns jsonb
language plpgsql
security definer
set search_path=public
as $$
declare
  v_me uuid:=auth.uid();

  v_region text:=lower(coalesce(p_region,'global'));
  v_country text:=upper(
    nullif(btrim(coalesce(p_country_code,'')),'')
  );
  v_period text:=lower(coalesce(p_period,'daily'));
  v_game text:=lower(coalesce(p_game_id,'all'));
  v_metric text:=lower(coalesce(p_metric,'score'));

  v_today date:=(now() at time zone 'UTC')::date;
  v_week date:=date_trunc(
    'week',
    (now() at time zone 'UTC')
  )::date;
  v_month date:=date_trunc(
    'month',
    (now() at time zone 'UTC')
  )::date;

  v_period_type text;
  v_period_start date;

  v_limit integer:=least(
    100,
    greatest(10,coalesce(p_limit,100))
  );

  v_rows jsonb;
  v_user jsonb;
  v_my_groups jsonb;
  v_total integer;
begin
  if v_me is null then
    raise exception 'Authentication required';
  end if;

  if v_region not in ('global','country') then
    raise exception 'Invalid group ranking region';
  end if;

  if v_period not in ('daily','weekly','monthly') then
    raise exception 'Invalid group ranking period';
  end if;

  if v_metric not in ('score','streak') then
    raise exception 'Invalid group ranking metric';
  end if;

  if v_region='country'
     and (v_country is null or v_country !~ '^[A-Z]{2}$') then

    return jsonb_build_object(
      'rows','[]'::jsonb,
      'user',null,
      'my_groups','[]'::jsonb,
      'total_players',0,
      'metric_label',case
        when v_metric='streak' then 'Group streak'
        when v_game='all' then 'Group Brain Score'
        when v_game in (
          'brainmix','flagdash','maphunt','brainiword'
        ) then 'Group Daily points'
        else 'Group points'
      end,
      'country_required',true
    );
  end if;

  if v_period='daily' then
    v_period_type:='day';
    v_period_start:=v_today;
  elsif v_period='weekly' then
    v_period_type:='week';
    v_period_start:=v_week;
  else
    v_period_type:='month';
    v_period_start:=v_month;
  end if;

  with group_base as (
    select
      g.id,
      g.name,
      g.country_code,
      g.crest_icon,
      g.crest_color,

      (
        select count(*)::integer
        from public.group_members gm
        where gm.group_id=g.id
      ) as member_count,

      exists(
        select 1
        from public.group_members mine
        where mine.group_id=g.id
          and mine.user_id=v_me
      ) as is_mine

    from public.groups g

    where g.status='active'
      and (
        v_region='global'
        or g.country_code=v_country
      )
      and not exists(
        select 1
        from public.admin_ranking_suspensions ars
        where ars.entity_type='group'
          and ars.entity_id=g.id
          and ars.active=true
          and (
            ars.expires_at is null
            or ars.expires_at>now()
          )
      )
  ),
  scored as (
    select
      gb.*,

      case
        when v_metric='streak'
          then public.brainilab_group_current_streak(
            gb.id
          )::bigint

        when v_game='all'
             and v_period='daily'
          then coalesce(gds.group_score,0)

        when v_game='all'
             and v_period in ('weekly','monthly')
          then coalesce(gps.group_score,0)

        else coalesce(ggps.group_score,0)
      end as score

    from group_base gb

    left join public.group_daily_stats gds
      on gds.group_id=gb.id
     and gds.stat_date=v_today

    left join public.group_period_stats gps
      on gps.group_id=gb.id
     and gps.period_type=case
       when v_period='weekly' then 'week'
       else 'month'
     end
     and gps.period_start=case
       when v_period='weekly' then v_week
       else v_month
     end

    left join public.group_game_period_stats ggps
      on ggps.group_id=gb.id
     and ggps.game_id=v_game
     and ggps.period_type=v_period_type
     and ggps.period_start=v_period_start

    where gb.member_count>=3
  ),
  eligible as (
    select *
    from scored
    where score>0
  ),
  ranked as (
    select
      s.*,
      row_number() over(
        order by
          s.score desc,
          lower(s.name),
          s.id
      ) as rank
    from eligible s
  )
  select
    coalesce(
      jsonb_agg(
        jsonb_build_object(
          'rank',r.rank,
          'group_id',r.id,
          'name',r.name,
          'country',r.country_code,

          'crest',jsonb_build_object(
            'icon',r.crest_icon,
            'color',r.crest_color
          ),

          'members',r.member_count,
          'score',r.score,

          'streak',case
            when v_metric='streak' then r.score
            else public.brainilab_group_current_streak(
              r.id
            )
          end,

          'display_value',case
            when v_metric='streak'
              then r.score::text || ' days'
            else to_char(
              r.score,
              'FM999G999G999G990'
            )
          end,

          'is_me',r.is_mine
        )
        order by r.rank
      ) filter(where r.rank<=v_limit),
      '[]'::jsonb
    ),

    count(*)::integer,

    (
      select jsonb_build_object(
        'rank',mine.rank,
        'group_id',mine.id,
        'name',mine.name,
        'country',mine.country_code,

        'crest',jsonb_build_object(
          'icon',mine.crest_icon,
          'color',mine.crest_color
        ),

        'members',mine.member_count,
        'score',mine.score,

        'streak',case
          when v_metric='streak' then mine.score
          else public.brainilab_group_current_streak(
            mine.id
          )
        end,

        'display_value',case
          when v_metric='streak'
            then mine.score::text || ' days'
          else to_char(
            mine.score,
            'FM999G999G999G990'
          )
        end,

        'is_me',true
      )
      from ranked mine
      where mine.is_mine=true
      order by mine.rank
      limit 1
    )

  into v_rows,v_total,v_user
  from ranked r;


  select coalesce(
    jsonb_agg(
      jsonb_build_object(
        'group_id',g.id,
        'name',g.name,
        'country',g.country_code,

        'crest',jsonb_build_object(
          'icon',g.crest_icon,
          'color',g.crest_color
        ),

        'members',(
          select count(*)
          from public.group_members cgm
          where cgm.group_id=g.id
        ),

        'eligible',(
          select count(*)>=3
          from public.group_members cgm
          where cgm.group_id=g.id
        )
      )
      order by g.created_at desc
    ),
    '[]'::jsonb
  )
  into v_my_groups
  from public.group_members gm
  join public.groups g
    on g.id=gm.group_id
   and g.status='active'
  where gm.user_id=v_me;


  return jsonb_build_object(
    'rows',v_rows,
    'user',v_user,
    'my_groups',v_my_groups,
    'total_players',v_total,

    'metric_label',case
      when v_metric='streak' then 'Group streak'
      when v_game='all' then 'Group Brain Score'
      when v_game in (
        'brainmix','flagdash','maphunt','brainiword'
      ) then 'Group Daily points'
      else 'Group points'
    end,

    'period',v_period,
    'game_id',v_game,
    'region',v_region,
    'country',v_country,
    'generated_at',now()
  );
end;
$$;
